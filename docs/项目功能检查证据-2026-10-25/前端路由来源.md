function Protected() {
  const navigate = useNavigate();
  const cache = useQueryClient();
  const q = useQuery({
    queryKey: ['session'],
    queryFn: authApi.me,
    retry: false,
  });
  useEffect(() => {
    const refreshPlanning = () => {
      const keys = new Set([
        'annualBudgets',
        'annualHistory',
        'savingsGoals',
        'goalDetail',
        'financialIndicators',
        'financialSummary',
        'cashflow',
        'cashflowPlans',
        'carryovers',
        'budgetPlanningHistory',
        'budgetAlerts',
        'budgetAlertSettings',

        'debt-plans',
        'debt-plan',
        'financial-health',
        'metric-details',
        'investments',
        'investment-trades',
        'investment-valuations',
      ]);
      void cache.invalidateQueries({
        predicate: (query) => keys.has(String(query.queryKey[0])),
      });
    };
    const logout = () => {
      void cache.cancelQueries();
      cache.clear();
      useUI.getState().closeEntry();
      navigate('/login', { replace: true });
    };
    window.addEventListener('finance:unauthorized', logout);
    window.addEventListener('finance:planning-changed', refreshPlanning);
    return () => {
      window.removeEventListener('finance:unauthorized', logout);
      window.removeEventListener('finance:planning-changed', refreshPlanning);
    };
  }, [navigate, cache]);
  if (q.isPending) return <Skeleton active />;
  if (q.isError)
    return (q.error as { code?: string }).code === 'UNAUTHORIZED' ? (
      <Navigate to="/login" replace />
    ) : (
      <ErrorState error={q.error} retry={() => q.refetch()} />
    );
  return <Outlet />;
}

export function canonicalNavigationPath(path: string) {
  const alias = Object.keys(navigationAliases)
    .sort((a, b) => b.length - a.length)
    .find((key) => path === key || path.startsWith(key + '/'));
  return alias ? navigationAliases[alias] + path.slice(alias.length) : path;
}
export function activeNavigation(path: string) {
  path = canonicalNavigationPath(path);
  if (path === '/admin' || path.startsWith('/admin/'))
    return {
      key: '/admin',
      label: tr('后台管理'),
      group: 'settings',
      groupLabel: settingsNavigation.label,
    };
  const routes = [...navigationGroups, settingsNavigation].flatMap((g) =>
    g.routes.map(([key, label]) => ({
      key,
      label,
      group: g.key,
      groupLabel: g.label,
    })),
  );
  return routes
    .filter(
      (r) => path === r.key || (r.key !== '/' && path.startsWith(r.key + '/')),
    )
    .sort((a, b) => b.key.length - a.key.length)[0];
}
